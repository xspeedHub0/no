local function enc(s)
    local out, kl = {}, #XKEY
    for i = 1, #s do
        out[i] = string.format("%02x", bit32.bxor(string.byte(s, i), string.byte(XKEY, ((i - 1) % kl) + 1)))
    end
    return table.concat(out)
end
local function dec(h)
    if type(h) ~= "string" or #h < 2 then return nil end
    local out, kl, j = {}, #XKEY, 0
    for i = 1, #h, 2 do
        j = j + 1
        local b = tonumber(h:sub(i, i + 1), 16)
        if not b then return nil end
        out[j] = string.char(bit32.bxor(b, string.byte(XKEY, ((j - 1) % kl) + 1)))
    end
    return table.concat(out)
end

local UNIT_POW = { K = 3, M = 6, B = 9, T = 12, QA = 15, QI = 18, SX = 21, SP = 24, OC = 27, NO = 30, DC = 33 }
local function parseCash(text)
    if type(text) ~= "string" then return nil end
    local raw, unit = text:match("^%s*%$?%s*([%d%.,]+)%s*(%a*)")
    local clean = raw and raw:gsub(",", "") or nil
    local num = clean and tonumber(clean)
    if not num then return nil end
    local decimals = clean:match("%.(%d+)$")
    local step = 10 ^ -(decimals and #decimals or 0)
    unit = unit:upper()
    if unit ~= "" then
        local pw = UNIT_POW[unit]
        if not pw then return nil end
        num = num * 10 ^ pw
        step = step * 10 ^ pw
    end
    return num, step
end

local CASH_UNITS = { "K", "M", "B", "T", "Qa", "Qi", "Sx", "Sp", "Oc", "No", "Dc" }
local function gameCash(g)
    g = math.floor((tonumber(g) or 0) + 0.5)
    if type(NumberUtils) ~= "table" then pcall(loadDuelModules) end
    if type(NumberUtils) == "table" and type(NumberUtils.ToString) == "function" then
        local ok, out = pcall(NumberUtils.ToString, NumberUtils, g)
        if ok and type(out) == "string" and out ~= "" then return out end
    end
    local unit, n = "", g
    for _, u in ipairs(CASH_UNITS) do
        if n < 1000 then break end
        n, unit = n / 1000, u
    end
    if unit == "" then return tostring(n) end
    local s = string.format("%.1f", math.floor(n * 10 + 1e-6) / 10):gsub("%.0$", "")
    return s .. unit
end

local function sameCash(text, g)
    if type(text) ~= "string" then return false end
    local core = text:gsub("%s", ""):gsub("^%$", ""):gsub("/s$", "")
    return core ~= "" and core == gameCash(g)
end

local function solveMutation(ti, traitSum, traits)
    local hint = ti.hookMut
    if hint then return hint, true end
    local row = ti.real and type(AnimalsData) == "table" and AnimalsData[ti.real] or nil
    local base = type(row) == "table" and tonumber(row.Generation) or nil
    if not (base and base > 0 and parseCash(ti.cashReal)) then return nil, false end
    if sameCash(ti.cashReal, base * formulaFor(nil, traitSum, traits)) then return nil, true end
    gameData()
    local found
    for k, d in pairs(type(MutData) == "table" and MutData or {}) do
        if type(d) == "table" and tonumber(d.Modifier)
            and sameCash(ti.cashReal, base * formulaFor(k, traitSum, traits)) then
            if found and mutModifier(found) ~= tonumber(d.Modifier) then return nil, false end
            found = found or k
        end
    end
    if found then return found, true end
    return nil, false
end

local function refreshMult(sp, ti)
    local traits, tsum = traitsOf(sp)
    ti.traits, ti.traitSum = traits, tsum
    local mut, known = solveMutation(ti, tsum, traits)
    if known then ti.mutation = mut end
    local formula = formulaFor(ti.mutation, tsum, traits)
    local row = ti.real and type(AnimalsData) == "table" and AnimalsData[ti.real] or nil
    local base = type(row) == "table" and tonumber(row.Generation) or nil
    local shown, step = parseCash(ti.cashReal)
    if base and base > 0 and shown and shown > 0 then
        if sameCash(ti.cashReal, base * formula) then
            ti.mult = formula
        else
            local mid = shown + ((step or 0) > 1 and step / 2 or 0)
            ti.mult = mid / base
        end
    else
        ti.mult = ti.mult or formula
    end
    return ti.mult
end

local function realFromTitle(text)

    if first and mset[norml(first)] then
        idx = animalKeyOf(rest)
        if idx then return idx, first end
    end
    local two, rest2 = text:match("^(%S+%s+%S+)%s+(.+)$")
    if two and mset[norml(two)] then
        idx = animalKeyOf(rest2)
        if idx then return idx, two end
    end
    return nil, nil
end

    end

    for _, sp in ipairs(tiles("Other")) do
        local ti = infoFor(sp)
        local title, cash = sp:FindFirstChild("Title"), sp:FindFirstChild("Cash")
        local vp = sp:FindFirstChild("ViewportFrame")

        local gi, gargs = itemOf(sp)
        if gi then
            ti.real, ti.hookMut = gi, mutationOf(gargs)
        else
            local mi = vp and modelIndexOf(vp) or nil
            if mi then
                ti.real = mi
            elseif not ti.real and title and title:IsA("TextLabel") and not labels[title] then
                local real = realFromTitle(title.Text)
                if real and (real ~= TD.key or next(swaps) == nil) then ti.real = real end
            end
        end
        ti.same = ti.real ~= nil and ti.real == TD.key

        if ti.same then
            if title then unforce(title) end
            if cash then unforce(cash) end
            if vp and drawn[vp] then
                drawn[vp] = nil
                if fakeVps[vp] then fakeVps[vp] = nil; redraw(vp, TD.key, false, gargs) end
            end
        else
            if ti.real then TD.real[ti.real] = true; swapIn(ti.real) end
            if cash and cash:IsA("TextLabel") and not labels[cash] then ti.cashReal = cash.Text end
            refreshMult(sp, ti)

            if title and title:IsA("TextLabel") then
                force(title, fakeNameFor(ti, info), function(realText)
                    local real = realFromTitle(realText)
                    if real and real ~= TD.key and not swaps[real] then ti.real = real end
                    return fakeNameFor(ti, currentInfo() or info)
                end)
                local pr = ti.real and pristineRow(ti.real)
                if pr and pr.DisplayName and labels[title] then labels[title].orig = pr.DisplayName end
            end
            if cash and cash:IsA("TextLabel") then
                force(cash, genText(info, ti.mult), function(realText)
                    if parseCash(realText) then ti.cashReal = realText end
                    return genText(currentInfo() or info, refreshMult(sp, ti))
                end)
            end
            local d = vp and drawn[vp]
            if vp and not d then
                local keep = fakeVps[vp]
                drawn[vp] = { real = ti.real, args = gargs, mut = ti.mutation }
                redraw(vp, TD.key, keep, argsFor(vp, ti))
                fakeVps[vp] = true
            elseif vp and not d.byGame and not gargs and d.mut ~= ti.mutation then
                d.mut = ti.mutation
                redraw(vp, TD.key, true, argsFor(vp, ti))
            elseif vp then
                local waiting = pendingDraw[vp] and os.clock() < pendingDraw[vp]
                local hasFake = false
                local fakes = {}
                for _, ch in ipairs(vp:GetChildren()) do
                    if isModelObj(ch) then
                        local inner = ch:IsA("WorldModel") and ch:FindFirstChildWhichIsA("Model") or ch
                        local nm = inner and inner.Name
                        if nm == TD.key then
                            fakes[#fakes + 1] = ch
                            hasFake = true
                        elseif nm ~= nil and (nm == ti.real or (not waiting and AnimalsData[nm] ~= nil)) then
                            pcall(function() ch:Destroy() end)
                        end
                    end
                end
                if #fakes > 1 and not waiting then
                    for i = 1, #fakes - 1 do pcall(function() fakes[i]:Destroy() end) end
                end
                if not hasFake and not waiting then
                    redraw(vp, TD.key, false, argsFor(vp, ti))
                end
            end
        end
    end
end

local BASE_ICON = "rbxassetid://139326264265904"
local BASE_GREY = Color3.new(0.65098, 0.65098, 0.65098)
local popProps = {}
local function popProp(inst, prop, value)
    if typeof(inst) ~= "Instance" then return end
    local rec = popProps[inst]
    if not rec then rec = {}; popProps[inst] = rec end
    local r = rec[prop]
    if not r then r = { orig = inst[prop] }; rec[prop] = r end
    r.val = value
    if inst[prop] ~= value then pcall(function() inst[prop] = value end) end
end

local function popRelease()
    for label, rec in pairs(popText) do
        if label.Parent and label.Text == rec.text then pcall(function() label.Text = rec.orig end) end
    end
    popText = {}
    for inst, props in pairs(popProps) do
        for prop, r in pairs(props) do
            if inst.Parent and inst[prop] == r.val then pcall(function() inst[prop] = r.orig end) end
        end
    end
    popProps = {}
    if popExists then
        if popExists.frame.Parent then pcall(function() popExists.frame.Visible = popExists.visible end) end
        popExists = nil
    end
end

local function applyPopupNow()
    if not (TD.on and TD.key) or type(AnimalsData) ~= "table" then return false end
    local p = popupParts(true)
    local tile = p and hoveredTheirs() or nil
    if not tile then return false end
    local info = AnimalsData[TD.key]
    if type(info) ~= "table" then return false end
    local ti = infoFor(tile)
    if ti.same then return false end

    popSet(p.name, fakeNameFor(ti, info))
    if not ti.mult then refreshMult(tile, ti) end
    popSet(p.cps, genText(info, ti.mult))
    if info.Rarity then popSet(p.rarity, tostring(info.Rarity)) end
    popSet(p.obtained, obtainedText(info))

    if p.exists then
        if not popExists then popExists = { frame = p.exists, visible = p.exists.Visible } end
        local mut = ti.mutation
        local ed = existData()
        local rec = ed and ed[TD.key]
        if type(rec) == "table" and type(rec.mutations) == "table" then
            local n = tonumber(rec.mutations[mut or "None"]) or 0
            local lbl = p.exists:FindFirstChild("Label")
            if lbl then popSet(lbl, existText(n) .. " Exist") end
            gameData()
            local md = mut and type(MutData) == "table" and MutData[mut] or nil
            local icon = p.exists:FindFirstChild("ITEM_IMAGE")
            popProp(lbl, "TextColor3", (type(md) == "table" and typeof(md.MainColor) == "Color3") and md.MainColor or BASE_GREY)
            popProp(icon, "Image", (type(md) == "table" and type(md.Icon) == "string") and md.Icon or BASE_ICON)
            popProp(icon, "ImageColor3", Color3.new(1, 1, 1))
            local animated = type(md) == "table" and md.GradientPreset ~= nil
            for _, d in ipairs(p.exists:GetDescendants()) do
                if d:IsA("UIGradient") then
                    if animated then
                        local rec = popProps[d] and popProps[d].Enabled
                        if rec then pcall(function() d.Enabled = rec.orig end); popProps[d].Enabled = nil end
                    else
                        popProp(d, "Enabled", false)
                    end
                end
            end
            p.exists.Visible = true
        else
            p.exists.Visible = false
        end
    end
    return true
end

local popWatch, popBusy = {}, false
local function onPopupWrite()
    if popBusy then return end
    popBusy = true
    pcall(applyPopupNow)
    popBusy = false
end

local function popupWatch()
    local p = popupParts(true)
    if not p then return end
    local targets = {
        { p.name, "Text" }, { p.rarity, "Text" }, { p.cps, "Text" },
        { p.obtained, "Text" }, { p.exists, "Visible" },
    }
    local lbl = p.exists and p.exists:FindFirstChild("Label")
    if lbl then targets[#targets + 1] = { lbl, "Text" }; targets[#targets + 1] = { lbl, "TextColor3" } end
    local icon = p.exists and p.exists:FindFirstChild("ITEM_IMAGE")
    if icon then
        targets[#targets + 1] = { icon, "Image" }
        targets[#targets + 1] = { icon, "ImageColor3" }
    end
    if p.exists then targets[#targets + 1] = { p.exists, "DescendantAdded" } end
    for _, t in ipairs(targets) do
        local inst, prop = t[1], t[2]
        local key = tostring(prop) .. ":" .. tostring(inst)
        if typeof(inst) == "Instance" and not popWatch[inst] then popWatch[inst] = {} end
        local conns = typeof(inst) == "Instance" and popWatch[inst] or nil
        if conns and not conns[prop] then
            if prop == "DescendantAdded" then
                conns[prop] = inst.DescendantAdded:Connect(onPopupWrite)
            else
                conns[prop] = inst:GetPropertyChangedSignal(prop):Connect(onPopupWrite)
            end
        end
    end
end

local function popUnwatch()
    for inst, conns in pairs(popWatch) do
        for prop, conn in pairs(conns) do pcall(function() conn:Disconnect() end) end
    end
    popWatch = {}
end

local function popupTick()
    if not (TD.on and TD.key) then
        if next(popText) or popExists then popRelease() end
        return
    end
    popupWatch()
    if not applyPopupNow() and (next(popText) or popExists) then popRelease() end
end

    if type(index) ~= "string" or index == "" then return index end
    local mine = underSide(vp, "Your")

    if not (mine or theirs) then return index end

    local args = table.pack(...)
    seen[vp] = { index = index, args = args }

    if not (TD.on and TD.key) then return index end
    if mine then
        TD.mine[index] = true
        if swaps[index] then swapOut(index) end
        return index
    end
    TD.real[index] = true

        local ti = infoFor(sp)
        ti.real, ti.hookMut = index, mutationOf(args)

    pendingDraw[vp] = os.clock() + 1.5
    fakeVps[vp] = true
    swapIn(index)
    queuePatch()

local function bforce(inst, prop, fn)
    local per = bLabels[inst]
    if not per then per = {}; bLabels[inst] = per end
    local w = per[prop]
    if not w then
        w = { orig = inst[prop] }
        per[prop] = w
        w.conn = inst:GetPropertyChangedSignal(prop):Connect(function()
            if w.dead then return end
            local cur = inst[prop]
            if w.text ~= nil and cur == w.text then return end
            w.orig = cur
            local ok, res = pcall(w.fn, cur)
            if ok and res ~= nil then
                w.text = res
                if cur ~= res then pcall(function() inst[prop] = res end) end
            else
                w.text = nil
            end
        end)
    end
    w.fn = fn
    local ok, res = pcall(fn, w.orig)
    if ok and res ~= nil then
        w.text = res
        if inst[prop] ~= res then pcall(function() inst[prop] = res end) end
    end
end

local function xzDist(a, b)
    local dx, dz = a.X - b.X, a.Z - b.Z

end

local function recGen(rec, cur)
    for _, m in ipairs({ rec.mult, formulaFor(rec.mut, rec.traitSum, rec.traits), formulaFor(rec.mut, 0, {}) }) do
        if m and sameCash(cur, rec.rGen * m) then
            rec.mult = m
            return "$" .. gameCash(rec.dGen * m) .. "/s"
        end
    end
    local v, step = parseCash(cur)
    if not v or rec.rGen <= 0 then return nil end
    local mid = v + (((step or 0) > 1) and step / 2 or 0)
    return "$" .. gameCash(rec.dGen * mid / rec.rGen) .. "/s"
end

local function hideReal(rec)
    rec.hidden = rec.hidden or {}
    local h = rec.hidden
    for _, obj in ipairs(rec.model:GetDescendants()) do
        if obj:IsA("BasePart") then
            if h[obj] == nil then h[obj] = { ltm = obj.LocalTransparencyModifier, trans = obj.Transparency } end
            if obj.LocalTransparencyModifier ~= 1 then pcall(function() obj.LocalTransparencyModifier = 1 end) end
            if obj.Transparency ~= 1 then pcall(function() obj.Transparency = 1 end) end
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            if h[obj] == nil then h[obj] = { trans = obj.Transparency } end
            if obj.Transparency ~= 1 then pcall(function() obj.Transparency = 1 end) end
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Highlight")
            or obj:IsA("Light") or obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") then
            if h[obj] == nil then h[obj] = { enabled = obj.Enabled } end
            if obj.Enabled then pcall(function() obj.Enabled = false end) end
        end
    end
end

local function unhideReal(rec)
    for obj, v in pairs(rec.hidden or {}) do
        if obj.Parent then
            pcall(function()
                if obj:IsA("BasePart") then
                    obj.LocalTransparencyModifier = v.ltm
                    obj.Transparency = v.trans
                elseif obj:IsA("Decal") or obj:IsA("Texture") then
                    obj.Transparency = v.trans
                else
                    obj.Enabled = v.enabled
                end
            end)
        end
    end
    rec.hidden = {}
end

local function attachBack(rec)
    local sh = rec.shift
    if sh then
        rec.shift = nil
        if sh.part.Parent then pcall(function() sh.part.CFrame = sh.cf end) end
    end
    local a = rec.att
    if not a then return end
    rec.att = nil
    if a.parent and a.parent.Parent and a.parent:IsDescendantOf(workspace) then
        pcall(function() a.inst.Parent = a.parent; a.inst.CFrame = a.cf end)
    else
        pcall(function() a.inst:Destroy() end)
    end
end

local function attachToClone(rec)
    if rec.att or rec.shift or not (rec.clone and rec.model) then return end
    if rec.key == "Headless Horseman" then return end
    local ra = rec.model:FindFirstChild("OVERHEAD_ATTACHMENT", true)
    if not (ra and ra:IsA("Attachment")) then return end
    local parent, cf
    local ca = rec.clone:FindFirstChild("OVERHEAD_ATTACHMENT", true)
    if ca and ca:IsA("Attachment") then
        parent, cf = ca.Parent, ca.CFrame
    else
        local root = rec.clone.PrimaryPart or rec.clone:FindFirstChildWhichIsA("BasePart")
        if not root then return end
        local ext = rec.clone:GetExtentsSize()
        local mod = tonumber(rowOf(rec.key).OverheadYOffsetModifier) or 0
        parent = root
        cf = root.CFrame:ToObjectSpace(rec.clone:GetPivot() * CFrame.new(0, ext.Y * 0.75 + mod, 0))
    end
    rec.att = { inst = ra, parent = ra.Parent, cf = ra.CFrame }
    pcall(function() ra.Parent = parent; ra.CFrame = cf end)
end

local function dropClone(rec)
    attachBack(rec)

    rec.cleanups = {}
    if rec.clone then pcall(function() rec.clone:Destroy() end) end
    rec.clone = nil
end

local function makeClone(rec, at, weldTo)
    local get = Disguise.origGetModel or (type(BrainrotAssets) == "table" and BrainrotAssets.getModel)
    if type(get) ~= "function" then return nil end
    local ok, src = pcall(get, rec.key)
    if not ok or typeof(src) ~= "Instance" then return nil end
    local c = safe(function() return src:Clone() end)
    if not c then return nil end
    c.Name = "__TDGBase"
    local root = c.PrimaryPart or c:FindFirstChild("RootPart") or c:FindFirstChildWhichIsA("BasePart")
    local rigged = c:FindFirstChildWhichIsA("Motor6D", true) ~= nil
    for _, p in ipairs(c:GetDescendants()) do
        if p:IsA("BasePart") then
            pcall(function()
                p.CanCollide, p.CanQuery, p.CanTouch, p.Massless = false, false, false, true
                p.Anchored = (not weldTo) and ((not rigged) or (p == root))
            end)
        end
    end
    c.Parent = workspace
    pcall(function() c:PivotTo(at:GetPivot()) end)
    if weldTo then
        for _, p in ipairs(c:GetDescendants()) do
            if p:IsA("BasePart") and (p == root or not rigged) then
                local w = Instance.new("WeldConstraint")
                w.Part0, w.Part1 = weldTo, p
                w.Parent = p
            end
        end
    end
    local cleanups = {}
    baseMods()
    if type(AnimalsShared) == "table" then
        if rec.mut and type(AnimalsShared.ApplyMutation) == "function" then
            local okm, cl = pcall(AnimalsShared.ApplyMutation, AnimalsShared, c, rec.key, rec.mut)
            if okm and type(cl) == "function" then table.insert(cleanups, cl) end
        end
        if #rec.traits > 0 and type(AnimalsShared.ApplyTraits) == "function" then
            local okt, cl = pcall(AnimalsShared.ApplyTraits, AnimalsShared, c, rec.key, rec.traits)
            if okt and type(cl) == "function" then table.insert(cleanups, cl) end
        end
    end
    if rigged then
        pcall(function()
            local a = ReplicatedStorage:FindFirstChild("Animations")
            local f = a and a:FindFirstChild("Animals")
            local k = f and f:FindFirstChild(rec.key)
            local idle = k and k:FindFirstChild("Idle")
            if not (idle and idle:IsA("Animation")) then return end
            local ctrl = c:FindFirstChildWhichIsA("Humanoid") or c:FindFirstChildWhichIsA("AnimationController")
            if not ctrl then ctrl = Instance.new("AnimationController"); ctrl.Parent = c end
            local animator = ctrl:FindFirstChildWhichIsA("Animator")
            if not animator then animator = Instance.new("Animator"); animator.Parent = ctrl end
            local track = animator:LoadAnimation(idle)
            track.Looped = true
            track:Play()
        end)
    end
    return c, cleanups
end

local function buildClone(rec)
    local pbKey = rec.real .. "|" .. norml(rec.mut or "")
    local pb = Base.prebuilt[pbKey]
    if pb and pb.clone and pb.clone.Parent then
        Base.prebuilt[pbKey] = nil
        rec.clone, rec.cleanups = pb.clone, pb.cleanups
        rec.pivot = safe(function() return rec.model:GetPivot() end)
        pcall(function() rec.clone:PivotTo(rec.model:GetPivot()) end)
        return
    end
    local c, cl = makeClone(rec, rec.model, nil)
    if c then
        rec.clone, rec.cleanups, rec.pivot = c, cl, safe(function() return rec.model:GetPivot() end)
    end
end

local function isGhost(rec)
    for _, p in ipairs(rec.model:GetDescendants()) do
        if p:IsA("MeshPart") then
            return p.Transparency >= 0.3 and p.Transparency < 0.95
        end
    end
    return false
end

local function setCloneFade(rec, fade)
    if not rec.clone or rec.fade == fade then return end
    rec.fade = fade
    for _, p in ipairs(rec.clone:GetDescendants()) do
        if p:IsA("BasePart") then pcall(function() p.LocalTransparencyModifier = fade end) end
    end
end

local function carriedBy(m)
    local hrp = safe(function() return LP.Character.HumanoidRootPart end)
    local root = m:FindFirstChild("RootPart")
    if not (hrp and root) then return false end
    for _, j in ipairs(root:GetJoints()) do
        local other = (j.Part0 == root) and j.Part1 or j.Part0
        if other == hrp then return true end
        if other and other ~= root and not other:IsDescendantOf(m) then
            for _, j2 in ipairs(other:GetJoints()) do
                if j2.Part0 == hrp or j2.Part1 == hrp then return true end
            end
        end
    end
    return false
end

local function carryOff()
    local c = Base.carry
    if not c then return end
    Base.carry = nil
    for _, f in ipairs(c.cleanups or {}) do pcall(f) end
    if c.clone then pcall(function() c.clone:Destroy() end) end
    if c.model and c.model.Parent then unhideReal(c) end
end

local function carryCheck()
    if Base.nativeMove or LP:GetAttribute("Stealing") then return end
    local c = Base.carry
    if c and (not c.model.Parent or not c.rec.model or not isGhost(c.rec)) then carryOff(); c = nil end
    if c then hideReal(c); return end
    for _, rec in ipairs(Base.recs) do
        if rec.model and rec.model.Parent and isGhost(rec) then
            for _, m in ipairs(workspace:GetChildren()) do
                if m.Name == rec.real and m:IsA("Model") and carriedBy(m) then
                    local holder = { rec = rec, model = m, hidden = {} }
                    hideReal(holder)
                    local root = m:FindFirstChild("RootPart") or m.PrimaryPart
                    holder.clone, holder.cleanups = makeClone(rec, m, root)
                    Base.carry = holder
                    return
                end
            end
        end
    end
end

local function rowSwap(rec)
    if Base.rows[rec.real] or type(AnimalsData) ~= "table" then return end
    local row, src = AnimalsData[rec.real], pristineRow(rec.key)
    if type(row) ~= "table" or type(src) ~= "table" then return end
    Base.rows[rec.real] = true
    row.DisplayName, row.Rarity = src.DisplayName, src.Rarity
end

local function rowUnswap(real)
    if not Base.rows[real] then return end
    Base.rows[real] = nil
    local row, pr = type(AnimalsData) == "table" and AnimalsData[real] or nil, pristineRow(real)
    if type(row) == "table" and type(pr) == "table" then
        row.DisplayName, row.Rarity = pr.DisplayName, pr.Rarity
    end
end

local function sellLabel(lbl)
    if Base.sellLabels[lbl] or not lbl:IsA("TextLabel") or lbl.Name ~= "Description" then return end
    local function check()

        if type(text) ~= "string" or text:sub(1, 20) ~= "Do you want to sell " then return end
        local hrp = safe(function() return LP.Character.HumanoidRootPart end)
        local best, bd
        for _, rec in ipairs(Base.recs) do
            if text == "Do you want to sell " .. rec.rName .. "?" and rec.pos then
                local d = hrp and (hrp.Position - rec.pos).Magnitude or 0
                if not bd or d < bd then best, bd = rec, d end

            bforce(lbl, "Text", function(cur)
                if cur == "Do you want to sell " .. rec.rName .. "?" then return "Do you want to sell " .. rec.dName .. "?" end
                return nil
            end)
        end
    end
    Base.sellLabels[lbl] = lbl:GetPropertyChangedSignal("Text"):Connect(function() pcall(check) end)
    pcall(check)
end

local function unstyle(rec)
    for label, state in pairs(rec.grad or {}) do
        if state.strokeAdded then state.strokeAdded:Disconnect() end
        if state.clean then pcall(state.clean) end
        if state.copy then state.copy:Destroy() end
        for stroke in pairs(state.strokes) do bunforce(stroke) end
        bunforce(label)
    end
    rec.grad = {}
end

local function styleRarity(rec, label)
    baseMods()
    rec.grad = rec.grad or {}
    if rec.grad[label] then return end
    local rarity = type(Rarities) == "table" and Rarities[rec.dRarity] or nil
    local copy = label:Clone()
    if not copy then return end
    for _, child in ipairs(copy:GetChildren()) do
        if not (child:IsA("UIStroke") or child:IsA("UIPadding") or child:IsA("UITextSizeConstraint")) then
            child:Destroy()
        end
    end
    for _, child in ipairs(copy:GetDescendants()) do
        if child:IsA("UIGradient") then child:Destroy() end
    end
    for _, stroke in ipairs(copy:GetChildren()) do
        if stroke:IsA("UIStroke") then
            stroke.Color, stroke.Transparency, stroke.Enabled = Color3.new(0, 0, 0), 0, true
        end
    end
    copy.Name = "__TDGPlacedRarity"
    copy.AnchorPoint, copy.Position, copy.Size = Vector2.zero, UDim2.fromOffset(0, 0), UDim2.fromScale(1, 1)
    copy.AutomaticSize, copy.Rotation = Enum.AutomaticSize.None, 0
    copy.BackgroundTransparency, copy.Visible = 1, true
    copy.ZIndex = label.ZIndex + 1
    copy.Text, copy.RichText, copy.TextTransparency = rec.dRarity, false, 0
    copy.TextColor3 = type(rarity) == "table" and typeof(rarity.Color) == "Color3" and rarity.Color or Color3.new(1, 1, 1)
    copy.TextStrokeColor3 = Color3.new(0, 0, 0)

    local state = {copy = copy, strokes = {}}
    rec.grad[label] = state
    local function hideNativeStroke(child)
        if child:IsA("UIStroke") and not state.strokes[child] then
            state.strokes[child] = true
            bforce(child, "Enabled", function() return false end)
        end
    end
    for _, child in ipairs(label:GetChildren()) do hideNativeStroke(child) end
    state.strokeAdded = label.ChildAdded:Connect(hideNativeStroke)
    bforce(label, "TextTransparency", function() return 1 end)
    bforce(label, "TextStrokeTransparency", function() return 1 end)
    copy.Parent = label

    local preset = type(rarity) == "table" and rarity.GradientPreset or nil
    if preset and type(Gradients) == "table" and type(Gradients.apply) == "function" then
        local ok, cleanup = pcall(Gradients.apply, copy, preset)
        if ok and type(cleanup) == "function" then state.clean = cleanup end
    end
    if rec.dRarity == "OG" and not state.clean and not copy:FindFirstChildOfClass("UIGradient") then
        local gradient = Instance.new("UIGradient")
        gradient.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 196, 45)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(255, 245, 151)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(210, 142, 14)),
        })
        gradient.Parent = copy
    end
end

local function attachDisguiseLabel(rec, nativePart, gui)
    local supported = rec.key == "Headless Horseman" or rec.key == "Skibidi Toilet"
        or rec.key == "Meowl" or rec.key == "John Pork" or rec.key == "Strawberry Elephant"
        or rec.key == "Dragon Cannelloni"
    if not supported or not (gui:IsA("SurfaceGui") or gui:IsA("BillboardGui")) then return end
    local originalAdornee = gui.Adornee
    local anchor = Instance.new("Part")
    anchor.Name = "__TDGDisguiseLabel"
    anchor.Anchored, anchor.CanCollide, anchor.CanTouch, anchor.CanQuery = true, false, false, false
    anchor.Transparency, anchor.CastShadow = 1, false
    anchor.Size, anchor.CFrame = nativePart.Size, nativePart.CFrame
    local displayGui = gui
    if gui:IsA("SurfaceGui") then
        displayGui = Instance.new("BillboardGui")
        displayGui.Name = "__TDGBillboardOverhead"
        displayGui.Size = UDim2.fromScale(nativePart.Size.X, nativePart.Size.Y)
        displayGui.AlwaysOnTop = gui.AlwaysOnTop
        displayGui.LightInfluence = 0
        displayGui.MaxDistance = gui.MaxDistance
        displayGui.ClipsDescendants = false
        for _, child in ipairs(gui:GetChildren()) do child.Parent = displayGui end
        displayGui.Parent = anchor
    end
    local lastClone, neck, stableHeight
    local fixedRotation = nativePart.CFrame.Rotation
    local function update()
        if displayGui ~= gui then displayGui.Enabled = gui.Enabled end
        local clone = rec.clone
        if not clone or not clone.Parent or not nativePart.Parent or nativePart.Position.Y > 5000 then
            anchor.CFrame = nativePart.CFrame
            return
        end
        if clone ~= lastClone then
            lastClone = clone
            local candidate = rec.key == "Headless Horseman" and clone:FindFirstChild("Torso.003", true)
            neck = candidate and candidate:IsA("Bone") and candidate or nil
            local point = clone:FindFirstChild("OVERHEAD_ATTACHMENT", true)
            local pivot = clone:GetPivot()
            if point and point:IsA("Attachment") then
                -- Read the authored height once; animated bones must not move the caption.
                stableHeight = point.WorldPosition.Y - pivot.Position.Y + 0.84
            else
                local bounds, size = clone:GetBoundingBox()
                stableHeight = bounds.Position.Y + size.Y * 0.5 - pivot.Position.Y + 1.5
            end
        end
        local position
        if neck and neck.Parent then
            position = neck.TransformedWorldCFrame.Position + Vector3.new(0, 3, 0)
        elseif rec.key == "Headless Horseman" then
            position = clone:GetPivot():PointToWorldSpace(Vector3.new(0, 12.15, -0.65))
        else
            position = clone:GetPivot().Position + Vector3.new(0, stableHeight, 0)
        end
        anchor.CFrame = CFrame.new(position) * (rec.key == "Headless Horseman" and nativePart.CFrame.Rotation or fixedRotation)
    end
    update()
    anchor.Parent = workspace
    displayGui.Adornee = anchor
    local follow = RunService.RenderStepped:Connect(update)
    rec.overheadCleanup = function()
        follow:Disconnect()
        if displayGui ~= gui then
            if gui.Parent then
                for _, child in ipairs(displayGui:GetChildren()) do child.Parent = gui end
            end
            displayGui:Destroy()
        elseif gui.Parent and gui.Adornee == anchor then gui.Adornee = originalAdornee end
        anchor:Destroy()
    end
end

local function overheadDistance(rec, h)
    local model = rec.model
    if not model or not model.Parent or not h.Parent then return end
    local attachment = model:FindFirstChild("OVERHEAD_ATTACHMENT", true)
    local expected = attachment and attachment:IsA("Attachment") and attachment.Parent
        and attachment.WorldPosition + Vector3.new(0, 0.84, 0)
    if not expected then
        local pivot = model:GetPivot().Position
        expected = Vector3.new(pivot.X, overheadY(model), pivot.Z)
    end
    local horizontal = xzDist(h.Position, expected)
    local vertical = math.abs(h.Position.Y - expected.Y)
    if horizontal <= 5 and vertical <= 7 then return horizontal * horizontal + vertical * vertical end
end

local function findOverhead(rec)
    local debris = workspace:FindFirstChild("Debris")
    local best, bd
    for _, h in ipairs(debris and debris:GetChildren() or {}) do
        local ao = h:IsA("BasePart") and h:FindFirstChild("AnimalOverhead")
        local dn = ao and ao:FindFirstChild("DisplayName", true)
        if dn and dn:IsA("TextLabel") and (Base.ov[h] == nil or Base.ov[h] == rec) then
            local per = bLabels[dn]
            local txt = (per and per.Text and per.Text.orig) or dn.Text
            if txt == rec.rName or txt == rec.dName then
                local d = overheadDistance(rec, h)
                if d and (not bd or d < bd) then best, bd = h, d end
            end
        end
    end
    return best
end

local function applyOverhead(rec, h)
    local ao = h:FindFirstChild("AnimalOverhead")
    if not ao or not overheadDistance(rec, h) then return end
    if rec.overhead then releaseOverhead(rec) end
    Base.ov[h] = rec
    rec.overhead, rec.ovLabels = h, {}
    local function lab(name)
        local l = ao:FindFirstChild(name, true)
        if l and l:IsA("TextLabel") then table.insert(rec.ovLabels, l); return l end
    end
    local dn, gen, price, rar = lab("DisplayName"), lab("Generation"), lab("Price"), lab("Rarity")
    if dn then
        bforce(dn, "Text", function(cur)
            if cur ~= rec.rName and cur ~= rec.dName then
                task.defer(function() if rec.overhead == h then releaseOverhead(rec) end end)
                return nil
            end
            return rec.dName
        end)
    end
    if gen then bforce(gen, "Text", function(cur) return recGen(rec, cur) end) end
    if price then
        bforce(price, "Text", function() return "$" .. gameCash(rec.dPrice) end)
    end
    if rar then styleRarity(rec, rar) end
    attachDisguiseLabel(rec, h, ao)
end

local function findPad(rec)
    local claim = rec.podium and safe(function() return rec.podium.Claim.Main end)
    if not claim then return nil end
    local debris = workspace:FindFirstChild("Debris")
    local best, bd
    for _, h in ipairs(debris and debris:GetChildren() or {}) do
        if h:IsA("BasePart") and h:FindFirstChild("CashPad") then
            local d = (h.Position - claim.Position).Magnitude
            if d < 5 and (not bd or d < bd) then best, bd = h, d end
        end
    end
    return best
end

local function applyPad(rec, h)
    local cp = h:FindFirstChild("CashPad")
    if not cp then return end
    rec.pad, rec.padLabels = h, {}
    local amount = cp:FindFirstChild("CollectAmount", true)
    local offline = cp:FindFirstChild("Offline", true)
    if amount and amount:IsA("TextLabel") then
        table.insert(rec.padLabels, amount)
        bforce(amount, "Text", function(cur)
            local v, step = parseCash(cur)
            if not v then return nil end
            if rec.padReal and v < rec.padReal * 0.5 then
                rec.collected = { v = rec.padReal, step = rec.padStep or 0, t = os.clock() }
            end
            rec.padReal, rec.padStep = v, step
            return "$" .. gameCash(v * rec.k)
        end)
    end
    if offline and offline:IsA("TextLabel") then
        table.insert(rec.padLabels, offline)
        bforce(offline, "Text", function(cur)
            return (cur:gsub("%$([%d%.,]+%a*)", function(s)
                local v = parseCash(s)
                return v and ("$" .. gameCash(v * rec.k)) or nil
            end))
        end)
    end
end

local function applyPrompts(rec)
    if not rec.podium then return end
    rec.prompts = rec.prompts or {}
    for pp in pairs(rec.prompts) do
        if not pp.Parent then rec.prompts[pp] = nil; Base.promptOwner[pp] = nil end
    end
    for _, pp in ipairs(rec.podium:GetDescendants()) do
        if pp:IsA("ProximityPrompt") and not rec.prompts[pp] then
            rec.prompts[pp] = true
            Base.promptOwner[pp] = rec
            promptEvents()
            bforce(pp, "ObjectText", function(cur)
                if cur == rec.rName or cur == rec.dName then return rec.dName end
                return nil
            end)
            bforce(pp, "ActionText", function(cur)
                local num = type(cur) == "string" and cur:match("^Sell:%s*%$(.+)$")
                local v = num and parseCash(num)
                if not v or rec.rPrice <= 0 then return nil end
                if num == cash2(rec.rPrice * 0.5) then return "Sell: $" .. cash2(rec.dPrice * 0.5) end
                return "Sell: $" .. cash2(v * rec.dPrice / rec.rPrice)
            end)
        end
    end
end

local function restoreRec(rec)
    releaseOverhead(rec)
    for _, l in ipairs(rec.padLabels or {}) do bunforce(l) end
    rec.pad, rec.padLabels = nil, nil
    releasePrompts(rec)
    if Base.carry and Base.carry.rec == rec then carryOff() end
    dropClone(rec)
    if rec.model then unhideReal(rec) end
    if rec.model then Base.owned[rec.model] = nil end
end

local function newRec(m, e, plot)
    local r, d = rowOf(e.real), rowOf(e.key)
    local rGen, dGen = tonumber(r.Generation) or 0, tonumber(d.Generation) or 0
    if rGen <= 0 or dGen <= 0 then return nil end
    local rec = {
        model = m, real = e.real, key = e.key, mut = e.mut, traits = e.traits or {}, traitSum = e.traitSum or 0,
        mult = e.mult or formulaFor(e.mut, e.traitSum, e.traits),
        rName = r.DisplayName or e.real, dName = d.DisplayName or e.key,
        dRarity = d.Rarity or "Common",
        rGen = rGen, dGen = dGen, k = dGen / rGen,
        rPrice = tonumber(r.Price) or 0, dPrice = tonumber(d.Price) or 0,
        grad = {}, hidden = {}, cleanups = {},
    }
    rec.pos = safe(function() return m:GetPivot().Position end)
    rec.podium = rec.pos and podiumOf(plot, rec.pos) or nil
    Base.owned[m] = rec
    return rec
end

local function cashoutLabel(lbl)
    local function try()
        if bLabels[lbl] then return true end
        local num = type(lbl.Text) == "string" and lbl.Text:match("^%+%$(.+)$")
        local v = num and parseCash(num)
        if not v then return false end
        local hrp = safe(function() return LP.Character.HumanoidRootPart end)
        local best, be
        for _, rec in ipairs(Base.recs) do
            local near = true
            if hrp and rec.pos then near = (hrp.Position - rec.pos).Magnitude < 30 end
            if near then
                local cands = {}
                if rec.collected and not rec.collected.used and os.clock() - rec.collected.t < 3 then cands[#cands + 1] = rec.collected end
                if rec.padReal then cands[#cands + 1] = { v = rec.padReal, step = rec.padStep or 0 } end
                for _, c in ipairs(cands) do
                    local err = math.abs(v - c.v)
                    if err <= math.max(c.step * 1.5, v * 0.05) and (not be or err < be) then best, be = rec, err end
                end
            end
        end
        if not best then return true end
        local rec = best
        if rec.collected and not rec.collected.used then
            rec.collected.used = true
            addCollected(rec, v)
        end
        bforce(lbl, "Text", function(cur)
            local n = type(cur) == "string" and cur:match("^%+%$(.+)$")
            local x = n and parseCash(n)
            return x and ("+$" .. cash2(x * rec.k)) or nil
        end)
        return true
    end
    if try() then return end
    local conn
    conn = lbl:GetPropertyChangedSignal("Text"):Connect(function()
        if try() then conn:Disconnect() end
    end)
    task.delay(2, function() pcall(function() conn:Disconnect() end) end)
end

    Base.cashoutFrame = f
    Base.cashoutConn = f.ChildAdded:Connect(function(ch)
        if ch:IsA("TextLabel") and #Base.recs > 0 then pcall(cashoutLabel, ch) end
    end)
end

    Base.prebuilt = {}
    if Base.cashoutConn then pcall(function() Base.cashoutConn:Disconnect() end) end

    for _, k in ipairs({ "ppShown", "ppHidden", "carryConn", "plotConn", "plotGone", "debrisConn", "sellConn" }) do
        if Base[k] then pcall(function() Base[k]:Disconnect() end); Base[k] = nil end

    for real in pairs(Base.rows) do rowUnswap(real) end

    for _, c in pairs(Base.sellLabels) do pcall(function() c:Disconnect() end) end

    for inst in pairs(bLabels) do bunforce(inst) end
end

    for _, sp in ipairs(tiles("Other")) do
        local ti = tileInfo[sp]
        if ti and ti.real and not ti.same and TD.key then
            local tr = {}
            for i, t in ipairs(ti.traits or {}) do tr[i] = t end
            list[#list + 1] = { real = ti.real, mut = ti.mutation, traits = tr, traitSum = ti.traitSum,
                                mult = ti.mult, key = TD.key }
        end
    end
    if #list > 0 or #tiles("Other") > 0 then
        local stillWanted = {}
        for _, e in ipairs(list) do stillWanted[e.real .. "|" .. norml(e.mut or "")] = true end
        for k, pb in pairs(Base.prebuilt) do
            if not stillWanted[k] then
                for _, f in ipairs(pb.cleanups or {}) do pcall(f) end
                pcall(function() pb.clone:Destroy() end)
                Base.prebuilt[k] = nil
            end
        end
        for _, e in ipairs(list) do
            local pbKey = e.real .. "|" .. norml(e.mut or "")
            if not Base.prebuilt[pbKey] then
                local snap = { key = e.key, mut = e.mut, traits = e.traits or {}, traitSum = e.traitSum or 0 }
                task.spawn(function()
                    local dummy = Instance.new("Model")
                    local dp = Instance.new("Part")
                    dp.Anchored, dp.CanCollide, dp.CanTouch, dp.CanQuery = true, false, false, false
                    dp.Transparency = 1
                    dp.CFrame = CFrame.new(0, -100000, 0)
                    dp.Parent = dummy
                    dummy.PrimaryPart = dp
                    dummy.Parent = workspace
                    local c, cl = makeClone(snap, dummy, nil)
                    dummy:Destroy()
                    if not c then return end
                    for _, p in ipairs(c:GetDescendants()) do
                        if p:IsA("BasePart") then pcall(function() p.LocalTransparencyModifier = 1 end) end
                    end
                    if Base.prebuilt[pbKey] == nil then
                        Base.prebuilt[pbKey] = { clone = c, cleanups = cl or {} }
                    else
                        for _, f in ipairs(cl or {}) do pcall(f) end
                        pcall(function() c:Destroy() end)
                    end
                end)
            end
        end
        Base.offer = list
    end
end

local function refreshRec(rec, plot)
    if not (rec.model and rec.model.Parent == plot) then
        rec.lost = rec.lost or os.clock()
        local found, fd
        for _, m in ipairs(plotAnimals(plot)) do
            local p = safe(function() return m:GetPivot().Position end)
            local born = Base.addedAt[m]
            if not Base.owned[m] and born and born > rec.lost - 3 and m.Name == rec.real
            and norml(modelMut(m) or "") == norml(rec.mut or "") and p then
                local dd = rec.pos and xzDist(p, rec.pos) or 0
                if not fd or dd < fd then found, fd = m, dd end
            end
        end
        if found then
            dropClone(rec); rec.hidden = {}
            releaseOverhead(rec)
            if rec.model then Base.owned[rec.model] = nil end
            rec.model, rec.lost, rec.fade, rec.bornAt = found, nil, nil, Base.addedAt[found]
            Base.owned[found] = rec
        else
            if os.clock() - rec.lost > 4 and not (Base.carry and Base.carry.rec == rec) then
                return false
            end
            return true
        end
    end
    rec.pos = safe(function() return rec.model:GetPivot().Position end) or rec.pos
    local pod = rec.pos and podiumOf(plot, rec.pos) or nil
    if pod ~= rec.podium then
        for _, l in ipairs(rec.padLabels or {}) do bunforce(l) end
        rec.pad, rec.padLabels, rec.padReal, rec.collected = nil, nil, nil, nil
        releasePrompts(rec)
        releaseOverhead(rec)
        rec.podium = pod
    end
    hideReal(rec)
    if not rec.clone or not rec.clone.Parent then pcall(buildClone, rec) end
    setCloneFade(rec, isGhost(rec) and 0.5 or 0)
    local pv = safe(function() return rec.model:GetPivot() end)
    if rec.clone and pv and rec.pivot and (pv.Position - rec.pivot.Position).Magnitude > 0.05 then
        rec.pivot = pv
        pcall(function() rec.clone:PivotTo(pv) end)
    end
    if not (rec.overhead and overheadDistance(rec, rec.overhead)) then
        if rec.overhead then releaseOverhead(rec) end
        local h = rec.pos and findOverhead(rec)
        if h then pcall(applyOverhead, rec, h) end
    end
    if not (rec.pad and rec.pad.Parent) then
        local h = findPad(rec)
        if h then pcall(applyPad, rec, h) end
    end
    pcall(applyPrompts, rec)
    return true
end

local function claimNew(m, plot)
    if Base.owned[m] or not (m:IsA("Model") and type(AnimalsData) == "table" and AnimalsData[m.Name]) then return nil end
    if tradeOpen() then return nil end
    local added = Base.addedAt[m]
    if not added or not Base.tradeStartedAt or added < Base.tradeStartedAt then return nil end
    if not podiumOf(plot, m:GetPivot().Position) then return nil end
    local list
    if tradeOpen() then
        list = Base.offer
    elseif #Base.pending > 0 and os.clock() <= Base.pendingUntil then
        list = Base.pending
    end
    if not list or (Base.known and Base.known[m]) then return nil end
    for i, e in ipairs(list) do
        if e.real == m.Name and norml(e.mut or "") == norml(modelMut(m) or "") then
            local rec = newRec(m, e, plot)
            if rec then
                rec.bornAt = Base.addedAt[m] or os.clock()
                table.remove(list, i)
                if Base.known then Base.known[m] = true end
                Base.recs[#Base.recs + 1] = rec
                cashoutWatch()
            end
            return rec
        end
    end
    return nil
end

local function watchPlot(plot)
    if Base.plotConnFor == plot then return end
    for _, k in ipairs({ "plotConn", "plotGone", "debrisConn" }) do
        if Base[k] then pcall(function() Base[k]:Disconnect() end); Base[k] = nil end
    end
    Base.plotConnFor = plot
    Base.plotConn = plot.ChildAdded:Connect(function(m)
        Base.addedAt[m] = os.clock()
        if Base.nativeMove then return end
        if not (TD.on and TD.key) then return end
        pcall(function()
            if m:IsA("Model") then
                local mMut = safe(function() return modelMut(m) end) or ""
                for _, list in ipairs({ Base.offer, Base.pending }) do
                    for _, e in ipairs(list or {}) do
                        if e.real == m.Name and norml(e.mut or "") == norml(mMut) then
                            for _, p in ipairs(m:GetDescendants()) do
                                if p:IsA("BasePart") then p.LocalTransparencyModifier = 1 end
                            end
                            local pbKey = e.real .. "|" .. norml(e.mut or "")
                            local pb = Base.prebuilt[pbKey]
                            if pb and pb.clone and pb.clone.Parent then
                                pcall(function() pb.clone:PivotTo(m:GetPivot()) end)
                            end
                            break
                        end
                    end
                end
            end
        end)
        pcall(function()
            local rec = claimNew(m, plot)
            if rec then refreshRec(rec, plot) end
            for _, r in ipairs(Base.recs) do
                if r ~= rec and not (r.model and r.model.Parent == plot) then refreshRec(r, plot) end
            end
        end)
    end)
    Base.plotGone = plot.ChildRemoved:Connect(function(m)
        if Base.nativeMove then return end
        local rec = Base.owned[m]
        if rec and TD.on then pcall(refreshRec, rec, plot) end
    end)
    local debris = workspace:FindFirstChild("Debris")
    if debris then
        Base.debrisConn = debris.ChildAdded:Connect(function(h)
            if not h:IsA("BasePart") then return end
            Base.ovAdded[h] = os.clock()
            if not (TD.on and TD.key) or not h:FindFirstChild("AnimalOverhead") then return end
            for _, rec in ipairs(Base.recs) do
                if not (rec.overhead and rec.overhead.Parent) and rec.model and rec.pos then
                    local found = findOverhead(rec)
                    if found then pcall(applyOverhead, rec, found) end
                end
            end
        end)
    end
end

local baseTick
baseTick = function(open)
    local plot = myPlot()
    if open and not Base.wasOpen then
        Base.tradeStartedAt = os.clock()
        Base.known = {}
        Base.offer = {}
        for _, m in ipairs(plot and plotAnimals(plot) or {}) do Base.known[m] = true end
    elseif not open and Base.wasOpen then
        Base.pending, Base.pendingUntil = Base.offer, os.clock() + 45
        Base.offer = {}
    end
    Base.wasOpen = open
    if not plot then return end
    watchPlot(plot)
    for m in pairs(Base.addedAt) do
        if not m.Parent then Base.addedAt[m] = nil end
    end
    for h in pairs(Base.ovAdded) do
        if not h.Parent then Base.ovAdded[h] = nil end
    end

    if #Base.pending > 0 and os.clock() > Base.pendingUntil then
        Base.pending, Base.known = {}, nil
    end
    if #Base.pending > 0 or open then
        for _, m in ipairs(plotAnimals(plot)) do
            if not (Base.known and Base.known[m]) and not Base.owned[m] then claimNew(m, plot) end
        end
    end
    if #Base.recs == 0 then return end
    cashoutWatch()

    for _, rec in ipairs(Base.recs) do
        local c = rec.collected
        if c and not c.used and os.clock() - c.t > 1.5 then
            c.used = true
            addCollected(rec, c.v)
        end
    end
    moneyShow()
    for i = #Base.recs, 1, -1 do
        local rec = Base.recs[i]
        local ok, keep = pcall(refreshRec, rec, plot)
        if ok and keep == false then
            pcall(restoreRec, rec)
            table.remove(Base.recs, i)
        end
    end
    pcall(carryCheck)
    if not Base.carryConn then
        Base.carryConn = workspace.ChildAdded:Connect(function(m)
            if #Base.recs == 0 or not m:IsA("Model") then return end
            pcall(carryCheck)
            task.delay(0.15, function() pcall(carryCheck) end)
        end)
    end
    for inst in pairs(bLabels) do
        if typeof(inst) ~= "Instance" or inst.Parent == nil then bLabels[inst] = nil end
    end
end

TD.baseAdopt = function(model, key, mut, traits)
    local plot = myPlot()
    if not (plot and model and model.Parent == plot) then return false end
    local k = animalKeyOf(key)
    if not k then return false end
    if Base.owned[model] then
        return Base.owned[model].key == k
    end
    local tr = traits or {}
    local sum = 0
    gameData()
    for _, t in ipairs(tr) do
        local d = type(TraitData) == "table" and TraitData[t] or nil
        sum = sum + (type(d) == "table" and tonumber(d.MultiplierModifier) or 0)
    end
    local rec = newRec(model, { real = model.Name, key = k, mut = mut, traits = tr, traitSum = sum }, plot)
    if not rec then return false end
    Base.recs[#Base.recs + 1] = rec
    baseTick(tradeOpen())
    return true
end

TD.adoptNearest = function(key)
    if LP:GetAttribute("Stealing") or Base.nativeMove then
        return false, "Place the held brainrot first."
    end
    local plot, root = myPlot(), LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not plot or not root then return false, "Your base is not loaded." end
    local nearest, distance
    for _, model in ipairs(plotAnimals(plot)) do
        if not Base.owned[model] then
            local position = model:GetPivot().Position
            local d = (position - root.Position).Magnitude
            if podiumOf(plot, position) and (not distance or d < distance) then
                nearest, distance = model, d
            end
        end
    end
    if not nearest or distance > 30 then return false, "Stand near an undisguised brainrot in your base." end
    local ok = TD.baseAdopt(nearest, key, modelMut(nearest), {})
    if not ok then return false, "Could not disguise this item." end
    return true
end

local function anchorCloneAssemblies(clone)
    local parts, joints = {}, {}
    for _, object in ipairs(clone:GetDescendants()) do
        if not object:IsDescendantOf(clone) then continue end
        if object:IsA("BasePart") then
            object.CanCollide, object.CanQuery, object.CanTouch, object.Massless = false, false, false, true
            parts[object] = object
        elseif object:IsA("JointInstance") or object:IsA("WeldConstraint") then
            local a, b = object.Part0, object.Part1
            if a and b and a:IsDescendantOf(clone) and b:IsDescendantOf(clone) then
                joints[#joints + 1] = object
            else
                object:Destroy()
            end
        elseif object:IsA("Constraint") or object:IsA("BodyMover") or object:IsA("LuaSourceContainer")
            or object:IsA("ProximityPrompt") or object:IsA("ClickDetector") then
            object:Destroy()
        end
    end
    local function leader(part)
        while parts[part] ~= part do part = parts[part] end
        return part
    end
    for _, joint in ipairs(joints) do
        parts[leader(joint.Part1)] = leader(joint.Part0)
    end
    local anchors = {}
    if parts[clone.PrimaryPart] then anchors[leader(clone.PrimaryPart)] = clone.PrimaryPart end
    for part in pairs(parts) do
        local group = leader(part)
        anchors[group] = anchors[group] or part
        part.Anchored = anchors[group] == part
    end
end

local function carryName(move)

end

local function bindCarryName(move, object, property)

    move.nameBindings = move.nameBindings or {}
    if move.nameBindings[object] then return end
    move.nameBindings[object] = true
    bforce(object, property, function() return name end)
end

local function clearNativeCarry(move)
    if move and move.pendingCarry then
        unhideReal(move.pendingCarry)
        move.pendingCarry = nil
    end
    if move and move.nameBindings then
        for object in pairs(move.nameBindings) do bunforce(object) end
        move.nameBindings = nil
    end
    local visual = move and move.carryVisual
    if not visual then return end
    move.carryVisual = nil
    for _, cleanup in ipairs(visual.cleanups) do pcall(cleanup) end
    visual.clone:Destroy()
    unhideReal(visual)
end

local function showNativeCarry(move)
    if move.pendingCarry then hideReal(move.pendingCarry) end
    if move.carryVisual then
        local model = move.carryVisual.model
        if model.Parent and carriedBy(model) then
            hideReal(move.carryVisual)
            return
        end
        clearNativeCarry(move)
    end
    if move.loadingCarry or not move.grabbed then return end
    local entry = move.grabbed
    local function find(container)
        for _, model in ipairs(container:GetChildren()) do
            if model:IsA("Model") and model.Name == entry.real and carriedBy(model) then return model end
        end
    end
    local source = find(workspace) or (LP.Character and find(LP.Character))
    if not source then return end
    if move.pendingCarry and move.pendingCarry.model ~= source then
        unhideReal(move.pendingCarry)
        move.pendingCarry = nil
    end
    local pending = move.pendingCarry or {model = source, hidden = {}}
    move.pendingCarry = pending
    hideReal(pending)
    move.loadingCarry = true
    local clone, cleanups = nil, {}
    local ok = pcall(function()
        local get = Disguise.origGetModel or BrainrotAssets.getModel
        local asset = get(entry.key)
        if typeof(asset) ~= "Instance" or not asset:IsA("Model") then return end
        clone = asset:Clone()
        if not clone then return end
        clone.Name = "__TDGNativeCarryVisual"
        baseMods()
        if entry.mut and type(AnimalsShared.ApplyMutation) == "function" then
            local cleanup = AnimalsShared:ApplyMutation(clone, entry.key, entry.mut)
            if type(cleanup) == "function" then cleanups[#cleanups + 1] = cleanup end
        end
        if entry.traits and #entry.traits > 0 and type(AnimalsShared.ApplyTraits) == "function" then
            local cleanup = AnimalsShared:ApplyTraits(clone, entry.key, entry.traits)
            if type(cleanup) == "function" then cleanups[#cleanups + 1] = cleanup end
        end
        anchorCloneAssemblies(clone)
        if Base.nativeMove ~= move or not TD.on or not LP:GetAttribute("Stealing")
            or not source.Parent or not carriedBy(source) then return end
        local pivot = source:GetPivot()
        clone:PivotTo(pivot)
        clone.Parent = workspace
        local animations = ReplicatedStorage:FindFirstChild("Animations")
        local animalAnimations = animations and animations:FindFirstChild("Animals")
        local realAnimations = animalAnimations and animalAnimations:FindFirstChild(entry.real)
        local disguiseAnimations = animalAnimations and animalAnimations:FindFirstChild(entry.key)
        local driver = clone:FindFirstChildWhichIsA("Humanoid", true)
            or clone:FindFirstChildWhichIsA("AnimationController", true)
        if not driver then driver = Instance.new("AnimationController"); driver.Parent = clone end
        local animator = driver:FindFirstChildOfClass("Animator")
        if not animator then animator = Instance.new("Animator"); animator.Parent = driver end
        local activeTrack, activeAnimation
        local function updateCarryAnimation()
            local nativeAnimator = source:FindFirstChildWhichIsA("Animator", true)
            local nativeTrack
            for _, track in ipairs(nativeAnimator and nativeAnimator:GetPlayingAnimationTracks() or {}) do
                if track.IsPlaying and (not nativeTrack or track.WeightCurrent > nativeTrack.WeightCurrent) then
                    nativeTrack = track
                end
            end
            local name = nativeTrack and nativeTrack.Animation and nativeTrack.Animation.Name
            if nativeTrack and nativeTrack.Animation and realAnimations then
                for _, animation in ipairs(realAnimations:GetChildren()) do
                    if animation:IsA("Animation") and animation.AnimationId == nativeTrack.Animation.AnimationId then
                        name = animation.Name
                        break
                    end
                end
            end
            local animation = disguiseAnimations and disguiseAnimations:FindFirstChild(name or "Idle")
            animation = animation or (disguiseAnimations and disguiseAnimations:FindFirstChild("Idle"))
            if not animation or not animation:IsA("Animation") then return end
            if activeAnimation ~= animation then
                local loaded, track = pcall(animator.LoadAnimation, animator, animation)
                if not loaded then return end
                if activeTrack then activeTrack:Stop(0); activeTrack:Destroy() end
                activeTrack, activeAnimation = track, animation
                track.Looped = true
                track:Play(0)
                if nativeTrack and nativeTrack.Length > 0 and track.Length > 0 then
                    track.TimePosition = nativeTrack.TimePosition / nativeTrack.Length * track.Length
                end
            end
            if activeTrack then activeTrack:AdjustSpeed(nativeTrack and nativeTrack.Speed or 1) end
        end
        updateCarryAnimation()
        cleanups[#cleanups + 1] = function()
            if activeTrack then activeTrack:Stop(0); activeTrack:Destroy(); activeTrack = nil end
        end
        local visual = {model = source, clone = clone, cleanups = cleanups, hidden = pending.hidden}
        move.pendingCarry = nil
        move.carryVisual = visual
        local overheads = ReplicatedStorage:FindFirstChild("Overheads")
        local template = overheads and overheads:FindFirstChild("AnimalOverhead")
        if template then
            local gui = template:Clone()
            local root = clone.PrimaryPart or clone:FindFirstChildWhichIsA("BasePart", true)
            local info = rowOf(entry.key)
            if gui and root and (gui:IsA("BillboardGui") or gui:IsA("SurfaceGui")) then
                gui.Name = "__TDGCarryOverhead"
                local function text(name, value)
                    local label = gui:FindFirstChild(name, true)
                    if label and label:IsA("TextLabel") then label.Text = value end
                    return label
                end
                text("DisplayName", info.DisplayName or entry.key)
                text("Generation", "$" .. gameCash((tonumber(info.Generation) or 0)
                    * (entry.mult or formulaFor(entry.mut, entry.traitSum, entry.traits))) .. "/s")
                text("Price", "$" .. gameCash(tonumber(info.Price) or 0))
                local rarity = text("Rarity", info.Rarity or "OG")
                local display = {key = entry.key, clone = clone, dRarity = info.Rarity or "OG", grad = {}}
                if rarity then styleRarity(display, rarity) end
                gui.Adornee, gui.Parent, gui.Enabled = root, root, true
                attachDisguiseLabel(display, root, gui)
                cleanups[#cleanups + 1] = function()
                    if display.overheadCleanup then display.overheadCleanup() end
                    unstyle(display)
                    gui:Destroy()
                end
            elseif gui then gui:Destroy() end
        end
        local function nameLabel(object)
            if object:IsA("TextLabel") and (object.Name == "DisplayName"
                or object.Name == "AnimalName" or object.Name == "BrainrotName") then
                bindCarryName(move, object, "Text")
            end
        end
        for _, object in ipairs(source:GetDescendants()) do nameLabel(object) end
        local captions = source.DescendantAdded:Connect(nameLabel)
        cleanups[#cleanups + 1] = function() captions:Disconnect() end
        hideReal(visual)
        local nextAnimationCheck = 0
        local follow = RunService.RenderStepped:Connect(function()
            if not source.Parent or not carriedBy(source) then
                if Base.nativeMove == move then return end
                clearNativeCarry(move); return
            end
            local current = source:GetPivot()
            if current ~= pivot then clone:PivotTo(current); pivot = current end
            if os.clock() >= nextAnimationCheck then
                nextAnimationCheck = os.clock() + 0.1
                updateCarryAnimation()
            end
        end)
        cleanups[#cleanups + 1] = function() follow:Disconnect() end
    end)
    move.loadingCarry = nil
    if not ok or not move.carryVisual then
        for _, cleanup in ipairs(cleanups) do pcall(cleanup) end
        if clone then clone:Destroy() end
    end
end

local function beginNativeMove()
    if Base.nativeMove or not TD.on or #Base.recs == 0 then return end
    local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    local saved, grabbed = {}, nil
    local plot = myPlot()
    local grab = Base.lastNativeGrab
    if grab and (grab.plot ~= plot or os.clock() - grab.time > 8) then grab = nil end
    for _, rec in ipairs(Base.recs) do
        if rec.model and rec.model.Parent then
            local position = rec.model:GetPivot().Position
            local entry = {model = rec.model, real = rec.real, key = rec.key, mut = rec.mut,
                traits = rec.traits, traitSum = rec.traitSum, mult = rec.mult, position = position,
                item = rec.model, record = rec}
            saved[#saved + 1] = entry
            if (grab and rec.podium == grab.podium) or (not grab and isGhost(rec)) then grabbed = entry end
        end
    end
    local move = {saved = saved, grabbed = grabbed, key = TD.key, plot = plot,
        origin = grab and grab.position or (grabbed and grabbed.position)}
    Base.lastNativeGrab = nil
    if grabbed then
        hideReal(grabbed.record)
        setCloneFade(grabbed.record, 0.5)
    end
    carryOff()
    Base.nativeMove = move
end

local function resumeNativeMove(move)
    local plot = myPlot()
    if plot ~= move.plot or TD.key ~= move.key then Base.nativeMove = nil; return end
    for _, entry in ipairs(move.saved) do
        if not entry.restored then
            local best
            local destination = entry.position
            if move.destination then
                if entry == move.grabbed then destination = move.destination
                elseif move.origin and (entry.position - move.destination).Magnitude < 4 then destination = move.origin end
            end
            local ambiguous = false
            for _, model in ipairs(plotAnimals(plot)) do
                if (not Base.owned[model] or Base.owned[model] == entry.record)
                    and model.Name == entry.real and norml(modelMut(model)) == norml(entry.mut)
                    and (model:GetPivot().Position - destination).Magnitude < 4 then
                    if best then ambiguous = true; break end
                    best = model
                end
            end
            if ambiguous then best = nil end
            if best then
                local rec = Base.owned[best]
                if not rec then
                    pcall(restoreRec, entry.record)
                    for i = #Base.recs, 1, -1 do
                        if Base.recs[i] == entry.record then table.remove(Base.recs, i); break end
                    end
                    rec = newRec(best, {real = entry.real, key = entry.key, mut = entry.mut,
                        traits = entry.traits or {}, traitSum = entry.traitSum or 0, mult = entry.mult}, plot)
                    if rec then Base.recs[#Base.recs + 1] = rec end
                end
                if rec then
                    if entry == move.grabbed then clearNativeCarry(move) end
                    refreshRec(rec, plot)
                    entry.restored = true
                end
            end
        end
    end
    local complete = true
    for _, entry in ipairs(move.saved) do if not entry.restored then complete = false end end
    if complete or os.clock() - move.finishedAt > 3 then
        clearNativeCarry(move)
        Base.nativeMove = nil
    end
end

nativeMoveConnections[#nativeMoveConnections + 1] = RunService.PreRender:Connect(function()
    local move = Base.nativeMove
    if not move then return end
    if LP:GetAttribute("Stealing") then
        showNativeCarry(move)
        return
    end
    move.finishedAt = move.finishedAt or os.clock()
    resumeNativeMove(move)
end)

local function observeNativeGrab(prompt, player)

local function isShownText(obj, stop)
    local txt = safe(function() return obj.Text end)
    if not txt or txt == "" then return false end
    local tt = safe(function() return obj.TextTransparency end)
    if tt and tt >= 0.99 then return false end
    local node = obj
    while node and node ~= stop do
        if node:IsA("GuiObject") then
            local vis = safe(function() return node.Visible end)
            if vis == false then return false end
        end
        node = node.Parent
    end
    return true
end

local function detectMutation(template)
    local mutation
    for _, obj in ipairs(template:GetDescendants()) do
        local attrs = safe(function() return obj:GetAttributes() end)
        if type(attrs) == "table" then
            for _, v in pairs(attrs) do
                local n = norml(v)
                if MUT_SET[n] then mutation = MUT_SET[n]; break end
            end
        end
        if not mutation and (obj:IsA("TextLabel") or obj:IsA("TextButton")) and isShownText(obj, template) then
            local n = norml(safe(function() return obj.Text end))
            if MUT_SET[n] then mutation = MUT_SET[n] end
        end
        if mutation then break end
    end
    return mutation
end

    local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local now = tick()
    posHistory[#posHistory + 1] = { cframe = root.CFrame, time = now }
    local hist = (mode == "carry" and Settings.carryLag.history) or Settings.normalLag.history or 0.27

local function currentSpeed()
    local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not root then return 0 end
    local v = root.AssemblyLinearVelocity
    return Vector3.new(v.X, 0, v.Z).Magnitude
end
local function meetsSpeedReq()
    local s = currentSpeed()
    if mode == "normal" then return s >= (tonumber(Settings.normalLag.speedMin) or 35) end
    if mode == "carry"  then return s >= (tonumber(Settings.carryLag.speedMin) or 17) end
    return false
end
local function doRubberband()
    local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local vel = root.AssemblyLinearVelocity
    if Vector3.new(vel.X, 0, vel.Z).Magnitude < 1 then return end
    local hist = (mode == "carry" and Settings.carryLag.history) or Settings.normalLag.history or 0.27
    local targetTime, best = tick() - hist, nil
    for i = 1, #posHistory do if posHistory[i].time >= targetTime then best = posHistory[i].cframe break end end
    if not best then return end
    root.CFrame = best
    root.AssemblyLinearVelocity = vel
end
local function stopLoop() if intervalThread then pcall(task.cancel, intervalThread); intervalThread = nil end end
local function startLoop()
    stopLoop()
    intervalThread = task.spawn(function()
        local startTime, iteration = tick(), 0
        while isActive do
            while isActive and not meetsSpeedReq() do task.wait(0.05) end
            if not isActive then break end
            iteration = iteration + 1
            local iv = (mode == "carry" and Settings.carryLag.interval) or Settings.normalLag.interval or 0.6
            local sleepT = (startTime + iteration * iv) - tick()
            if sleepT > 0 then task.wait(sleepT) end
            if isActive and meetsSpeedReq() then doRubberband() end
        end
    end)
end
local function setMode(newMode)
    if mode == newMode then return end
    mode = newMode
    if mode then isActive = true; startLoop() else isActive = false; stopLoop() end
end

        if os.clock() < CueMuteUntil and type(sound) == "string" and MUTED_SOUND_PATHS[sound] then return end

                    if inst[prop] ~= ov[prop] then inst[prop] = ov[prop] end
                end)
                return
            end
            if Freezer.allowRecolor[inst]
            and (prop == "BackgroundColor3" or prop == "Color") then
                pcall(function() saved = inst[prop] end)
                return
            end
            pcall(function()
                if inst[prop] ~= saved then inst[prop] = saved end
            end)
        end)
    end)
    if ok2 and conn then table.insert(Freezer.conns, conn) end
end

local function lockInstance(d)
    if isAnimated(d) then return end
    if Freezer.pressOnly[d] == true then
        lockProp(d, "Visible")
        lockProp(d, "Text")
        return
    end
    if d:IsA("GuiObject") then
        for _, p in ipairs(LOCKED_GUI) do lockProp(d, p) end
    elseif d:IsA("UIStroke") then
        for _, p in ipairs(LOCKED_STROKE) do lockProp(d, p) end
    elseif d:IsA("UIGradient") then
        lockProp(d, "Color"); lockProp(d, "Enabled")
    end
    if d:IsA("TextLabel") or d:IsA("TextButton") then
        lockProp(d, "TextColor3")
        lockProp(d, "Text")
        if isTimerText(safe(function() return d.Text end)) then
            pcall(function() d.Visible = false end)
            lockProp(d, "Visible")
        end
    end
end

function Freezer:Unfreeze()
    self.active = false
    if setAutoAccept then pcall(setAutoAccept, false) end
    if LocalPick then
        LocalPick.enabled = false
        LocalPick.suspended = false
        LocalPick.current = nil
        pcall(function() LocalPick:ClearOutlines() end)
        for template, cover in pairs(LocalPick.covers) do
            pcall(function() cover:Destroy() end)
            LocalPick.covers[template] = nil
        end
    end
    for _, c in ipairs(self.conns) do pcall(function() c:Disconnect() end) end
    self.conns = {}
    self.roots = {}
    self.known = {}
    self.savedParents = {}
    for _, cover in ipairs(self.ownCovers) do pcall(function() cover:Destroy() end) end
    self.ownCovers = {}
    self.pressOnly = {}
    self.allowRecolor = {}
    self.colorOverride = {}
end

local DEAD_BUTTON_WORDS = { "cancel", "decline", "reject", "leave", "back", "close" }

local function isDeadButton(inst)
    if not inst:IsA("GuiButton") then return false end
    local txt = safe(function() return inst.Text end)
    local name = safe(function() return inst.Name end) or ""
    local hay = ((type(txt) == "string" and txt or "") .. " " .. name):lower()
    for _, w in ipairs(DEAD_BUTTON_WORDS) do
        if hay:find(w, 1, true) then return true end
    end
    return false
end

local function scaleColor(c, f)
    return Color3.new(math.clamp(c.R * f, 0, 1), math.clamp(c.G * f, 0, 1), math.clamp(c.B * f, 0, 1))
end

local function attachHoverTint(cover, target)
    local base = safe(function() return target.BackgroundColor3 end)
    if not base then return end
    local function paint(f)
        pcall(function()
            local now = safe(function() return target.BackgroundColor3 end)
            if now and now ~= scaleColor(base, 0.9) and now ~= scaleColor(base, 0.8) and now ~= base then base = now end
            target.BackgroundColor3 = f and scaleColor(base, f) or base
        end)
    end
    table.insert(Freezer.conns, cover.MouseEnter:Connect(function() paint(0.9) end))
    table.insert(Freezer.conns, cover.MouseLeave:Connect(function() paint(nil) end))
    table.insert(Freezer.conns, cover.MouseButton1Down:Connect(function() paint(0.8) end))
    table.insert(Freezer.conns, cover.MouseButton1Up:Connect(function() paint(0.9) end))
end

local function neutralizeDeadButtons(root)
    local list = root:GetDescendants()
    table.insert(list, root)
    for _, d in ipairs(list) do
        if isDeadButton(d) and not d:FindFirstChild("ScannerCover") then
            Freezer.pressOnly[d] = true
            pcall(function()
                local cover = Instance.new("TextButton")
                cover.Name = "ScannerCover"
                cover.Size = UDim2.new(1, 0, 1, 0)
                cover.BackgroundTransparency = 1
                cover.Text = ""
                cover.AutoButtonColor = false
                cover.ZIndex = 50
                cover.Active = true
                cover.Selectable = false
                cover.Parent = d
                table.insert(Freezer.ownCovers, cover)
                attachHoverTint(cover, d)
            end)
        end
    end
end

function Freezer:Freeze(roots)
    self:Unfreeze()
    if typeof(roots) == "Instance" then roots = { roots } end
    if type(roots) ~= "table" or #roots == 0 then return false, 0 end

    self.active = true
    self.roots  = roots
    if setAutoAccept then pcall(setAutoAccept, true) end
    if LocalPick then LocalPick.enabled = true end
    installCueMute()
    local count = 0

    for _, root in ipairs(roots) do
        if root and root.Parent then
            count = count + 1
            neutralizeDeadButtons(root)

            local list = root:GetDescendants()
            table.insert(list, root)
            for _, d in ipairs(list) do
                self.known[d] = true
                self.savedParents[d] = d.Parent
                lockInstance(d)
            end

            local ok, added = pcall(function()
                return root.DescendantAdded:Connect(function(d)
                    if not Freezer.active or Freezer.restoring then return end
                    task.defer(function()
                        if not Freezer.active or not d.Parent then return end
                        if not Freezer.known[d] and d.Name ~= "ScannerCover" then
                            pcall(function() d:Destroy() end)
                        end
                    end)
                end)
            end)
            if ok and added then table.insert(self.conns, added) end

            local ok2, removing = pcall(function()
                return root.DescendantRemoving:Connect(function(d)
                    if not Freezer.active or Freezer.restoring or not Freezer.known[d] then return end
                    local parent = Freezer.savedParents[d]
                    task.defer(function()
                        if not Freezer.active or not parent or not parent.Parent then return end
                        pcall(function()
                            if d and not d.Parent then
                                Freezer.restoring = true
                                d.Parent = parent
                                Freezer.restoring = false
                            end
                        end)
                    end)
                end)
            end)
            if ok2 and removing then table.insert(self.conns, removing) end
        end
    end

    if LocalPick then pcall(function() LocalPick:Arm() end) end

    return count > 0, count
end

local function getDuelRoot(template)
    local node = template
    while node and node.Parent do
        if node.Parent:IsA("LayerCollector") then return node end
        node = node.Parent
    end
    return template and template:FindFirstAncestorWhichIsA("Frame") or nil
end

local function looksLikeOfferHeader(txt)
    if type(txt) ~= "string" or txt == "" then return false end
    return txt:lower():find("offer", 1, true) ~= nil
end

local function getFreezeRoots(template)
    local roots = {}
    local function add(inst)
        if not inst or not inst.Parent then return end
        if table.find(roots, inst) then return end
        for _, r in ipairs(roots) do
            if inst:IsDescendantOf(r) then return end
        end
        roots[#roots + 1] = inst
    end

    if template then
        local sg = template:FindFirstAncestorWhichIsA("LayerCollector")
        add(sg or getDuelRoot(template))
    end

    local pg = safe(function() return LP:FindFirstChild("PlayerGui") end)
    if pg then
        for _, sg in ipairs(pg:GetChildren()) do
            if sg:IsA("LayerCollector") then
                pcall(function()
                    for _, d in ipairs(sg:GetDescendants()) do
                        if (d:IsA("TextLabel") or d:IsA("TextButton"))
                        and looksLikeOfferHeader(safe(function() return d.Text end)) then
                            add(sg)
                            break
                        end
                    end
                end)
            end
        end
    end

    return roots
end

local function findClickable(root)
    if not root or not root.Parent then return nil end
    if root:IsA("GuiButton") and root.Name ~= "ScannerCover" then return root end
    for _, d in ipairs(root:GetDescendants()) do
        if d:IsA("GuiButton") and d.Name ~= "ScannerCover" then return d end
    end
    local p = root.Parent
    while p and not p:IsA("LayerCollector") do
        if p:IsA("GuiButton") then return p end
        p = p.Parent
    end
    return nil
end

local function clickGui(btn)
    if typeof(firesignal) == "function" then
        local ok = pcall(function()
            firesignal(btn.MouseButton1Down)
            task.wait(0.03)
            firesignal(btn.MouseButton1Click)
            firesignal(btn.MouseButton1Up)
            if btn.Activated then firesignal(btn.Activated) end
        end)
        if ok then return true, "firesignal" end
    end

    if typeof(getconnections) == "function" then
        local fired = false
        pcall(function()
            for _, sig in ipairs({ btn.MouseButton1Click, btn.Activated, btn.MouseButton1Down }) do
                for _, c in ipairs(getconnections(sig)) do
                    local fn = safe(function() return c.Function end)
                    if fn then fn(); fired = true
                    elseif c.Fire then c:Fire(); fired = true end
                end
                if fired then break end
            end
        end)
        if fired then return true, "getconnections" end
    end

    local okVim = pcall(function()
        local pos   = btn.AbsolutePosition + btn.AbsoluteSize / 2
        local inset = GuiService:GetGuiInset()
        local x, y  = pos.X + inset.X, pos.Y + inset.Y

        task.wait(0.05)
        vim:SendMouseButtonEvent(x, y, 0, false, game, 0)
        task.wait(0.05)
    end)
    if okVim then return true, "virtualinput" end

    return false, "none"
end

local SELECT_FILL    = Color3.new(0.0588235, 0.196078, 0.0588235)
local SELECT_BORDER  = Color3.new(0, 1, 0)
local DEFAULT_FILL   = Color3.new(0.137255, 0.176471, 0.196078)
local DEFAULT_BORDER = Color3.new(0, 0, 0)

local function getSpacer(template)
    if template.Name == "Spacer" then return template end
    local sp = template:FindFirstChild("Spacer")
    if sp and sp:IsA("GuiObject") then return sp end
    for _, d in ipairs(template:GetDescendants()) do
        if d.Name == "Spacer" and d:IsA("GuiObject") then return d end
    end
    return nil
end

local function isGreenish(c)
    if not c then return false end
    return c.G > c.R + 0.03 and c.G > c.B + 0.03
end

local function resetTile(template, saved)
    local spacer = saved and saved.spacer or getSpacer(template)
    if not spacer or not spacer.Parent then return end
    local stroke = (saved and saved.stroke) or spacer:FindFirstChildOfClass("UIStroke")
    Freezer.allowRecolor[spacer] = nil
    forceColor(spacer, "BackgroundColor3", (saved and saved.bg) or DEFAULT_FILL)
    if stroke and stroke.Parent then
        Freezer.allowRecolor[stroke] = nil
        forceColor(stroke, "Color", (saved and saved.strokeColor) or DEFAULT_BORDER)
    end
end

function LocalPick:ClearOutlines(keep)
    for template, saved in pairs(self.outlines) do
        if template ~= keep then
            pcall(function() resetTile(template, saved) end)
            self.outlines[template] = nil
        end
    end

    local scroll = getDuelScroll()
    if not scroll then return end

    local function inKeep(inst)
        if not keep then return false end
        return inst == keep or inst:IsDescendantOf(keep)
    end

    for _, d in ipairs(scroll:GetDescendants()) do
        if not inKeep(d) then
            if d:IsA("UIStroke") then
                if isGreenish(safe(function() return d.Color end))
                or (Freezer.colorOverride[d] and Freezer.colorOverride[d].Color) then
                    Freezer.allowRecolor[d] = nil
                    forceColor(d, "Color", DEFAULT_BORDER)
                end
            elseif d:IsA("GuiObject") then
                if isGreenish(safe(function() return d.BackgroundColor3 end))
                or (Freezer.colorOverride[d] and Freezer.colorOverride[d].BackgroundColor3) then
                    Freezer.allowRecolor[d] = nil
                    forceColor(d, "BackgroundColor3", DEFAULT_FILL)
                end
            end
        end
    end
end

function LocalPick:ShowOutline(template)
    if not template or not template.Parent then return end
    self.current = template
    self:ClearOutlines(template)

    local spacer = getSpacer(template)
    if not spacer then return end
    local stroke = spacer:FindFirstChildOfClass("UIStroke")

    if not self.outlines[template] then
        self.outlines[template] = {
            spacer = spacer, bg = DEFAULT_FILL,
            stroke = stroke, strokeColor = DEFAULT_BORDER,
        }
    end

    Freezer.allowRecolor[spacer] = true
    if stroke then Freezer.allowRecolor[stroke] = true end
    forceColor(spacer, "BackgroundColor3", SELECT_FILL)
    if stroke then forceColor(stroke, "Color", SELECT_BORDER) end
end

function LocalPick:EnforceSingleOutline()
    local keep = self.current
    if keep and not keep.Parent then keep = nil; self.current = nil end
    if not keep then
        local scroll = getDuelScroll()
        if scroll then
            for _, child in ipairs(scroll:GetChildren()) do
                if child:IsA("GuiObject") then
                    local spacer = getSpacer(child)
                    if spacer then
                        local st = spacer:FindFirstChildOfClass("UIStroke")
                        if isGreenish(safe(function() return spacer.BackgroundColor3 end))
                        or (st and isGreenish(safe(function() return st.Color end))) then
                            keep = child
                            break
                        end
                    end
                end
            end
        end
        self:ClearOutlines(keep)
        return
    end
    self:ClearOutlines(keep)
    local spacer = getSpacer(keep)
    if spacer then
        local stroke = spacer:FindFirstChildOfClass("UIStroke")
        Freezer.allowRecolor[spacer] = true
        forceColor(spacer, "BackgroundColor3", SELECT_FILL)
        if stroke then
            Freezer.allowRecolor[stroke] = true
            forceColor(stroke, "Color", SELECT_BORDER)
        end
    end
end

function LocalPick:SetCoversActive(on)
    for template, cover in pairs(self.covers) do
        if cover and cover.Parent then
            pcall(function() cover.Visible = on end)
        else
            self.covers[template] = nil
        end
    end
end

    if not LocalPick.enabled or LocalPick.suspended then return end
    LocalPick:ShowOutline(template)
    task.defer(function()
        if LocalPick.current == template then
            pcall(function() LocalPick:EnforceSingleOutline() end)
        end
    end)
    task.delay(0.1, function()
        if LocalPick.current == template then
            pcall(function() LocalPick:EnforceSingleOutline() end)
        end
    end)
end

function LocalPick:Cover(template)
    if self.covers[template] then return false end
    local ok, cover = pcall(function()

        c.Name = "ScannerCover"
        c.Size = UDim2.new(1, 0, 1, 0)
        c.BackgroundTransparency = 1
        c.Text = ""
        c.AutoButtonColor = false
        c.ZIndex = 50
        c.Active = true
        c.Selectable = false
        c.Parent = template

    end)
    if not ok or not cover then return false end
    self.covers[template] = cover
    table.insert(self.conns, cover.MouseButton1Click:Connect(function() pressed(template) end))

end

function LocalPick:Arm()
    if not self.enabled then return 0 end
    local scroll = getDuelScroll()
    if not scroll then return 0 end

    for template, cover in pairs(self.covers) do
        if not template.Parent or not cover.Parent then self.covers[template] = nil end
    end

    local n = 0
    for _, child in ipairs(scroll:GetChildren()) do
        if child:IsA("GuiObject")
        and not child:IsA("UIListLayout") and not child:IsA("UIGridLayout") then
            if self:Cover(child) then n = n + 1 end
        end
    end
    return n
end

local function posOf(obj)
  local cur, hops = obj, 0
  while cur and cur ~= workspace and cur ~= game and hops < 15 do
    hops = hops + 1
    if cur:IsA("BillboardGui") then
      cur = cur.Adornee or cur.Parent
    elseif cur:IsA("BasePart") then
      return cur.Position
    elseif cur:IsA("Attachment") then
      return cur.WorldPosition
    elseif cur:IsA("Model") then
      local ok, cf = pcall(function() return cur:GetPivot() end)
      if ok and cf then return cf.Position end
      cur = cur.Parent
    elseif cur:IsA("PlayerGui") or cur:IsA("Players") then
      return nil
    else
      cur = cur.Parent
    end
  end
  return nil
end

local function containsName(text, name)
  if #name < 3 then return false end
  local start = 1
  while true do
    local a, b = string.find(text, name, start, true)
    if not a then return false end
    local before = a > 1 and string.sub(text, a - 1, a - 1) or ""
    local after = string.sub(text, b + 1, b + 1)
    if not string.find(before, "%w") and not string.find(after, "%w") then return true end
    start = b + 1
  end
end

local function matchesMe(value)
  if value == nil then return false end
  if typeof(value) == "Instance" then return value == player end
  local s = tostring(value)
  return s == player.Name or s == tostring(player.UserId) or s == player.DisplayName
end

local function ownsPlotByValue(plot)
  for _, key in ipairs(OWNER_KEYS) do
    local ok, value = pcall(function() return plot:GetAttribute(key) end)
    if ok and matchesMe(value) then return true end
    local child = plot:FindFirstChild(key)
    if child and child:IsA("ValueBase") then
      local ok2, childValue = pcall(function() return child.Value end)
      if ok2 and matchesMe(childValue) then return true end
    end
  end
  return false
end

  return false
end

local function overheadName(gui)
  local hasIncome, fallback, fallbackY, named, known = false, nil, nil, nil, nil
  for _, d in ipairs(gui:GetDescendants()) do
    if d:IsA("TextLabel") then
      local text = cleanText(d.Text)
      if text ~= "" then
        if string.find(text, "/s", 1, true) then hasIncome = true end
        if not known then known = knownName(text) end
        if not named and NAME_LABELS[d.Name] then named = text end
        if not string.find(text, "$", 1, true) and not string.find(text, "/s", 1, true) and string.find(text, "%a") then
          local y = d.Position.Y.Scale * 1000 + d.Position.Y.Offset
          if not fallbackY or y < fallbackY then fallback, fallbackY = text, y end
        end
      end
    end
  end
  if known then return known end
  local looksLikeBrainrot = hasIncome or string.find(gui.Name, "Overhead") or string.find(gui.Name, "Animal")
  if not looksLikeBrainrot then return nil end
  local pick = named or fallback
  if pick and not string.find(pick, "%a") then return nil end
  return pick
end

local function sortFloor(floor)
  local allNumbered = true
  for _, item in ipairs(floor) do
    if not item.num then allNumbered = false end
  end
  if allNumbered then
    table.sort(floor, function(a, b) return a.num < b.num end)
    return
  end
  local minX, maxX, minZ, maxZ = math.huge, -math.huge, math.huge, -math.huge
  for _, item in ipairs(floor) do
    minX = math.min(minX, item.pos.X); maxX = math.max(maxX, item.pos.X)
    minZ = math.min(minZ, item.pos.Z); maxZ = math.max(maxZ, item.pos.Z)
  end
  if maxX - minX >= maxZ - minZ then
    table.sort(floor, function(a, b) return a.pos.X < b.pos.X end)
  else
    table.sort(floor, function(a, b) return a.pos.Z < b.pos.Z end)
  end
end

local function groupFloors(items)
  table.sort(items, function(a, b) return a.pos.Y < b.pos.Y end)
  local floors, current, lastY = {}, nil, nil
  for _, item in ipairs(items) do
    if not current or item.pos.Y - lastY > FLOOR_GAP then
      current = {}
      floors[#floors + 1] = current
    end
    current[#current + 1] = item
    lastY = item.pos.Y
  end
  for _, floor in ipairs(floors) do sortFloor(floor) end
  return floors
end

local GEN_LABELS = { Generation = true, Gen = true, Income = true, PerSecond = true, Cash = true }
local SUFFIX = { k = 1e3, m = 1e6, b = 1e9, t = 1e12, qa = 1e15, qi = 1e18, sx = 1e21, sp = 1e24, oc = 1e27, no = 1e30 }

local function parseGenText(text)
  local number, suffix = string.match(tostring(text or ""), "([%d%.,]+)%s*(%a*)")
  number = tonumber((string.gsub(number or "", ",", "")))
  if not number then return nil end
  local multiplier = SUFFIX[string.lower(suffix or "")]
  if suffix ~= "" and not multiplier then return nil end
  return number * (multiplier or 1)
end

          local value = lower(text)
          if value ~= "none" and value ~= "normal" and value ~= "default" then mutation = text end
        end
      end
    end
  end
  if not mutation then mutation = safe(function() return detectMutation(gui) end) end

  local traits = {}
  local known = loadTraitNames()
  pcall(function()
    for _, d in ipairs(gui:GetDescendants()) do

            elseif not known and label ~= "" and not string.find(lower(label), "^template") and not string.find(label, "^UI") then
              traits[#traits + 1] = label
            end
          end
        end
        break
      end
    end
  end)
  return { cash = named or anyGen, mutation = mutation, traits = traits }
end

local function attachOverheads(items)
  local overheads = collectDebrisOverheads()
  local used = {}

  local function assign(onlyWithMutation)
    local candidates = {}
    for _, item in ipairs(items) do
      if item.name and item.name ~= "" and not item.gui and item.model then
        local mutation = modelMutation(item.model)
        if (mutation ~= nil) == onlyWithMutation then
          local pos = posOf(item.model)
          if pos then
            for _, overhead in ipairs(overheads) do
              if norml(overhead.name) == norml(item.name) then
                local theirs = overheadMutation(overhead.gui)
                if not mutation or (theirs and lower(theirs) == lower(mutation)) then
                  candidates[#candidates + 1] = { item = item, overhead = overhead, dist = (overhead.pos - pos).Magnitude }
                end
              end
            end
          end
        end
      end
    end
    table.sort(candidates, function(a, b) return a.dist < b.dist end)
    for _, pair in ipairs(candidates) do
      if not pair.item.gui and not used[pair.overhead] then
        used[pair.overhead] = true
        pair.item.gui = pair.overhead.gui
      end
    end
  end

  assign(true)
  assign(false)
end

local function scanBase()
  loadKnown()
  local plot = findMyPlot()
  if not plot then return nil end
  local podiums = findPodiums(plot)
  local candidates = collectCandidates(plot)

  if #podiums > 0 then
    for _, c in ipairs(candidates) do
      local best, bestDist = nil, NEAR_PODIUM
      for _, podium in ipairs(podiums) do
        if not podium.name then
          local dy = c.pos.Y - podium.pos.Y
          if dy > -4 and dy < 14 then
            local dx, dz = c.pos.X - podium.pos.X, c.pos.Z - podium.pos.Z
            local dist = math.sqrt(dx * dx + dz * dz)
            if dist < bestDist then best, bestDist = podium, dist end
          end
        end
      end
      if best then best.name = c.name; best.gui = c.gui; best.model = c.model end
    end
  end

  local items = (#podiums > 0) and podiums or candidates
  if #items == 0 then return nil end

  attachOverheads(items)

  local result = {}
  for floorNumber, floor in ipairs(groupFloors(items)) do
    for spot, item in ipairs(floor) do
      if item.name and item.name ~= "" then
        local entry = { name = (string.gsub(item.name, "[|;]", " ")), floor = floorNumber, spot = spot }
        if item.gui then
          local ok, info = pcall(readOverhead, item.gui)
          if ok and info then
            entry.cash, entry.mutation, entry.traits = info.cash, info.mutation, info.traits
          end
        end
        entry.mutation = modelMutation(item.model) or entry.mutation
        result[#result + 1] = entry
      end
    end
  end
  return result
end

local function baseGenOf(name)
  if type(AnimalsData) ~= "table" then return nil end
  local info = AnimalsData[name]
  if type(info) ~= "table" then
    for _, value in pairs(AnimalsData) do
      if type(value) == "table" and value.DisplayName == name then info = value; break end
    end
  end
  return type(info) == "table" and tonumber(info.Generation) or nil
end

local scanCache, scanCachedAt = nil, -100

local function buildReport()
  local duelList = collectBrainrots()
  if V2.inDuelServer() then return duelList end

  if not teleporting and os.clock() - scanCachedAt >= 4 then
    scanCachedAt = os.clock()
    local ok, result = pcall(scanBase)
    scanCache = ok and result or nil
  end
  local spots = scanCache
  if type(spots) ~= "table" or #spots == 0 then return duelList end

  local fromDuelList = {}
  for _, entry in ipairs(duelList) do
    local key = norml(entry.name)
    fromDuelList[key] = fromDuelList[key] or {}
    table.insert(fromDuelList[key], entry)
  end

  local report = {}
  for _, spot in ipairs(spots) do
    local item = { name = spot.name, spot = spot.spot, floor = spot.floor, mutation = spot.mutation,
      traits = (spot.traits and #spot.traits > 0) and spot.traits or nil }
    local base = baseGenOf(spot.name)

    local shown = spot.cash and parseGenText(spot.cash)
    if shown and (not base or shown >= base * 0.9) then item.cash = spot.cash end

    local list = fromDuelList[norml(spot.name)]
    local fromList = list and table.remove(list, 1)
    if not item.cash and fromList then
      item.cash = fromList.cash
      item.mutation = item.mutation or fromList.mutation
    end

    if not item.cash and base then item.cash = "$" .. gameName(base) .. "/s" end
    item.cash = item.cash or ""
    report[#report + 1] = item
  end
  return report
end

V2.collect = function()
  local ok, result = pcall(buildReport)
  if ok and type(result) == "table" then return result end
  return collectBrainrots()
end
end)()

local function warnReportFailure()

local function isCurrent()

local function scheduleRetry()
    failureCount = math.min(failureCount + 1, 6)
    local base = math.min(30, 2 ^ (failureCount - 1))
    retryAt = tick() + math.min(30, base * (0.8 + math.random() * 0.4))
end

    if not isCurrentSocket(s) then return end

    local w = WebSocket or (syn and syn.websocket) or genv.WebSocket

local function bindSocketEvent(s, eventName, callback)
    local event = safe(function() return s[eventName] end)
    if not event then return nil end
    local connection = safe(function() return event:Connect(callback) end)
    if connection then eventConnections[#eventConnections + 1] = connection end
    return connection

wsSendTable = function(t)
    local s = socket
    if not isCurrentSocket(s) then return false end
    local ok = pcall(function() s:Send(enc(HttpService:JSONEncode(t))) end)
    if not ok then socketDead(s) end
    return ok and isCurrentSocket(s)

local function beginConnect()
    if not isCurrent() or wsLive or wsConnecting then return end
    wsConnecting = true
    connectGeneration = connectGeneration + 1
    local generation = connectGeneration
    connectStartedAt = tick()
    -- A yielding native connect cannot block the supervisor or inventory worker.
    task.spawn(function()
        local ok, s = pcall(function()
            local w = wsResolve()
            if not w then return nil end
            return w.connect(WS_URL)
        end)
        if not isCurrent() or generation ~= connectGeneration then
            if ok and s then closeSocket(s) end
            return
        end
        wsConnecting = false
        if not ok or not s then scheduleRetry(); return end
        socket, wsLive, registered = s, true, false
        registrationSentAt = tick()
        lastPingAt, lastPongAt = tick() - APP_PING_INTERVAL, tick()
        lastSig, lastSendAt, catalogSent = nil, 0, false

        local messageConn = bindSocketEvent(s, "OnMessage", function(msg)
            if not isCurrentSocket(s) or type(msg) ~= "string" then return end
            if msg == "xeno:pong:v1" then lastPongAt = tick(); return end
            local plain = dec(msg)
            if not plain then return end
            local data = safe(function() return HttpService:JSONDecode(plain) end)
            if type(data) ~= "table" then return end
            if data.settings then
                registered = true
                failureCount = 0
                pcall(applySettings, data.settings)
            end
            if data.commands then pcall(applyCommands, data.commands) end
        end)
        if not messageConn then socketDead(s); return end
        bindSocketEvent(s, "OnClose", function() socketDead(s) end)
        bindSocketEvent(s, "OnError", function() socketDead(s) end)
        if not isCurrentSocket(s) then return end
        -- Authenticate immediately; no metadata query or inventory collection here.
        local sent = wsSendTable({
            scriptKey = CONFIG.SCRIPT_KEY, robloxId = LP.UserId, name = LP.Name,
            displayName = LP.DisplayName, executor = executorName,
            game = gameName(), placeId = game.PlaceId, jobId = game.JobId
        })
        if not sent then socketDead(s) end
    end)
end
local function payloadSig(p)
    local parts = { tostring(p.placeId), tostring(p.jobId), tostring(p.inDuel), tostring(p.tpBlocked) }
    for _, nm in ipairs(p.serverPlayers or {}) do parts[#parts + 1] = tostring(nm) end
    parts[#parts + 1] = "|"
    for _, b in ipairs(p.brainrots or {}) do
        parts[#parts + 1] = tostring(b.name) .. ":" .. tostring(b.gen or b.cash or "")
            .. ":" .. tostring(b.mutation or "")
            .. ":" .. tostring(b.floor or "") .. "." .. tostring(b.spot or "")
    end
    return table.concat(parts, ",")
end
wsSend = function(force)
    local s = socket
    if not isCurrentSocket(s) or not registered then return false end
    local built, p = pcall(payload)
    if not isCurrentSocket(s) then return false end
    if not built or type(p) ~= "table" then warnReportFailure(); return false end
    local signed, sig = pcall(payloadSig, p)
    if not isCurrentSocket(s) then return false end
    if not signed then warnReportFailure(); return false end
    if not force then
        if sig == lastSig then return true end
        if tick() - lastSendAt < MIN_GAP then return true end
    end
    local ok = wsSendTable(p)
    if ok then lastSig, lastSendAt = sig, tick() end
    return ok
end
local function supervise()
    if kicked then session.Stop(); return end
    local now = tick()
    if not wsLive then
        if wsConnecting and now - connectStartedAt >= CONNECT_TIMEOUT then
            -- Invalidate late results; their socket is closed when the call returns.
            connectGeneration = connectGeneration + 1
            wsConnecting = false
            scheduleRetry()
        end
        if not wsConnecting and now >= retryAt then beginConnect() end
        return
    end
    local s = socket
    if not registered and now - registrationSentAt >= REGISTER_TIMEOUT then
        socketDead(s)
    elseif now - lastPongAt > PONG_TIMEOUT then
        socketDead(s)
    elseif now - lastPingAt >= APP_PING_INTERVAL then
        lastPingAt = now
        local ok = pcall(function() s:Send("xeno:ping:v1") end)
        if not ok then socketDead(s) end
    end
    if registered and isCurrentSocket(s) and not gameInfoCache.loading and now >= gameInfoCache.retryAt then
        task.spawn(function() pcall(refreshGameInfo, isCurrent) end)
    end
end
